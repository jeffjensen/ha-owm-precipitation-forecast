=== Chart Visualizations

==== ApexCharts - Hourly Rain Forecast

[source,yaml]
----
type: custom:apexcharts-card
header:
  title: Hourly Rain Forecast
  show: true
  show_states: true
  colorize_states: true
graph_span: 24h
span:
  start: hour
series:
  - entity: sensor.owm_precipitation_forecast_home_next24h_rain
    name: Rain
    type: column
    data_generator: |
      return entity.attributes.hourly_breakdown.map(h => {
        return [new Date(h.time).getTime(), h.rain];
      });
    show:
      legend_value: false
yaxis:
  - min: 0
    apex_config:
      tickAmount: 4
      labels:
        formatter: |
          EVAL:function(value) {
            return value.toFixed(2) + ' in';
          }
----

== Automations

=== Precipitation Alerts

==== Heavy Rain Warning

[source,yaml]
----
automation:
  - alias: "Heavy Rain Alert"
    description: "Alert when significant rain is forecast"
    trigger:
      - platform: numeric_state
        entity_id: sensor.owm_precipitation_forecast_home_next24h_rain
        above: 1.5  # More than 1.5 inches
    condition:
      - condition: time
        after: "07:00:00"
        before: "22:00:00"  # Only during waking hours
    action:
      - service: notify.mobile_app
        data:
          title: "⛈️ Heavy Rain Expected"
          message: >
            {{ states('sensor.owm_precipitation_forecast_home_next24h_rain') }}
            inches of rain forecast in the next 24 hours.
            Consider:
            - Checking drainage
            - Moving vehicles
            - Securing outdoor items
          data:
            priority: high
            ttl: 0
----

          end_date_time: "{{ now().replace(hour=8, minute=0) + timedelta(days=1) }}"

=== Automated Actions

==== Close Windows When Rain Expected

[source,yaml]
----
automation:
  - alias: "Auto Close Windows - Rain Expected"
    description: "Close windows when rain is forecast"
    trigger:
      - platform: numeric_state
        entity_id: sensor.owm_precipitation_forecast_home_hourly_rain
        above: 0.1  # Any measurable rain
    condition:
      - condition: state
        entity_id: binary_sensor.bedroom_window
        state: "on"  # Window is open
      - condition: state
        entity_id: input_boolean.vacation_mode
        state: "off"  # Not on vacation
    action:
      - service: cover.close_cover
        target:
          entity_id: cover.bedroom_window
      
      - service: notify.mobile_app
        data:
          message: "Closed bedroom window - rain expected in next hour"
----

=== Daily/Weekly Reports

==== Morning Weather Brief

[source,yaml]
----
automation:
  - alias: "Morning Weather Brief"
    description: "Daily precipitation forecast summary"
    trigger:
      - platform: time
        at: "07:00:00"
    action:
      - service: notify.mobile_app
        data:
          title: "☀️ Morning Weather Brief"
          message: >
            {% set rain = states('sensor.owm_precipitation_forecast_home_daily_rain') | float %}
            {% set snow = states('sensor.owm_precipitation_forecast_home_daily_snow') | float %}
            
            **Today's Forecast:**
            
            {% if rain > 0 %}
            🌧️ Rain: {{ rain }} inches
            {% endif %}
            
            {% if snow > 0 %}
            ❄️ Snow: {{ snow }} inches
            {% endif %}
            
            {% if rain == 0 and snow == 0 %}
            ☀️ No precipitation expected
            {% endif %}
            
            Have a great day!
----

==== Weekly Precipitation Summary

[source,yaml]
----
automation:
  - alias: "Weekly Precipitation Summary"
    description: "Sunday evening forecast summary"
    trigger:
      - platform: time
        at: "19:00:00"
    condition:
      - condition: time
        weekday: sun
    action:
      - service: persistent_notification.create
        data:
          title: "📊 Week Ahead Forecast"
          message: >
            {% set daily_rain = state_attr('sensor.owm_precipitation_forecast_home_daily_rain', 'daily_breakdown') %}
            {% set daily_snow = state_attr('sensor.owm_precipitation_forecast_home_daily_snow', 'daily_breakdown') %}
            
            **7-Day Precipitation Forecast:**
            
            {% for day in daily_rain[:7] %}
            {{ (day.date | as_datetime).strftime('%A') }}: 
            {{ day.rain }}in rain, 
            {{ daily_snow[loop.index0].snow }}in snow
            {% endfor %}
----

== Template Sensors

=== Custom Calculations

==== Precipitation Likelihood Sensor

[source,yaml]
----
template:
  - sensor:
      - name: "Precipitation Likely Today"
        unique_id: precipitation_likely_today
        state: >
          {% set rain = states('sensor.owm_precipitation_forecast_home_daily_rain') | float %}
          {% set snow = states('sensor.owm_precipitation_forecast_home_daily_snow') | float %}
          {% if rain > 0.1 or snow > 0.5 %}
            Yes
          {% else %}
            No
          {% endif %}
        icon: >
          {% set rain = states('sensor.owm_precipitation_forecast_home_daily_rain') | float %}
          {% set snow = states('sensor.owm_precipitation_forecast_home_daily_snow') | float %}
          {% if snow > 0.5 %}
            mdi:weather-snowy
          {% elif rain > 0.1 %}
            mdi:weather-rainy
          {% else %}
            mdi:weather-sunny
          {% endif %}
----

==== Total Precipitation Sensor

[source,yaml]
----
template:
  - sensor:
      - name: "Total Precipitation Today"
        unique_id: total_precipitation_today
        unit_of_measurement: "in"
        state: >
          {% set rain = states('sensor.owm_precipitation_forecast_home_daily_rain') | float %}
          {% set snow_liquid = states('sensor.owm_precipitation_forecast_home_daily_snow') | float / 10 %}
          {{ (rain + snow_liquid) | round(2) }}
----

== Scripts

=== Manual Refresh Script

[source,yaml]
----
script:
  refresh_precipitation_forecast:
    alias: "Refresh Precipitation Forecast"
    icon: mdi:refresh
    sequence:
      - service: owm_precipitation_forecast.update_forecast
      - delay:
          seconds: 5
      - service: notify.mobile_app
        data:
          message: "Precipitation forecast updated"
----

        event_type: mobile_app_notification_action
        event_data:

          entity_id: all
