== Running Tests

=== Basic Test Execution

[source,bash]
----
# Run all tests
pytest

# Run with verbose output
pytest -v

# Run specific test file
pytest tests/test_owm_client.py

# Run specific test
pytest tests/test_sensor.py -v

pytest tests/test_owm_client.py::test_client_success

# Run tests matching pattern
pytest -k "test_client"
----

=== Coverage Reports

[source,bash]
----
# Generate coverage report
pytest --cov=custom_components/ha_owm_precipitation_forecast

# Generate HTML coverage report
pytest --cov --cov-report=html

# View HTML report
open htmlcov/index.html

# Generate XML coverage (for CI)
pytest --cov --cov-report=xml

# Show missing lines
pytest --cov --cov-report=term-missing
----

== Writing Tests

=== Test Structure

[source,python]
----
import pytest
from custom_components.ha_owm_precipitation_forecast.owm_client import OWMClient


async def test_descriptive_name():
    """Test description of what is being tested."""
    # Arrange - Set up test data and mocks
    client = OWMClient("test_key", 40.7128, -74.0060)

    # Act - Perform the action being tested
    result = await client.async_get_forecast()

    # Assert - Verify the results
    assert result is not None
    assert "hourly" in result

    # Cleanup
    await client.async_close()
----

=== Using Fixtures

Fixtures provide reusable test data:

[source,python]
----
@pytest.fixture
def mock_forecast_data():
    """Return mock forecast data."""
    return {
        "hourly": [...],
        "daily": [...]
    }


async def test_with_fixture(mock_forecast_data):
    """Test using fixture data."""
    assert "hourly" in mock_forecast_data
----

=== Mocking API Calls

[source,python]
----
from aioresponses import aioresponses

@pytest.mark.asyncio
async def test_api_call():
    """Test API call with mock."""
    with aioresponses() as mock:
        mock.get(
            "https://api.openweathermap.org/data/3.0/onecall",
            payload={"data": "test"},
            status=200
        )

        client = OWMClient("test_key", 40.0, -74.0)
        result = await client.async_get_forecast()

        assert result == {"data": "test"}
        await client.async_close()
----

=== Testing Error Conditions

[source,python]
----
@pytest.mark.asyncio
async def test_api_error():
    """Test API error handling."""
    with aioresponses() as mock:
        mock.get(OWM_API_URL, status=401)

        client = OWMClient("invalid_key", 40.0, -74.0)

        with pytest.raises(OWMClientError) as exc_info:
            await client.async_get_forecast()

        assert exc_info.value.error_code == "invalid_api_key"
        await client.async_close()
----

== Advanced Testing

=== Parameterized Tests

[source,python]
----
@pytest.mark.parametrize("temp,expected_ratio", [
    (35.0, 0.0),   # Above freezing
    (30.0, 10.0),  # Near freezing
    (25.0, 15.0),  # Cold
    (8.0, 40.0),   # Very cold
])
def test_snow_ratios(temp, expected_ratio):
    """Test snow calculation at various temperatures."""
    calc = PrecipitationCalculator()
    snow = calc.calculate_snow_from_rain(1.0, temp)
    assert snow == expected_ratio
----
